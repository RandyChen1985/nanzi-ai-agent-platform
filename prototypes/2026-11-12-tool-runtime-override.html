def resolve_tool_timeout(spec, tool_cfg, agent_timeout, global_timeout):
    # 1) 工具级配置（本原型新增，1–3600）
    if (v := _coerce_positive(tool_cfg)) is not None: return v, "tool"
    # 2) 工具自带超时（脚本工具注册时带入，修掉被抹平的缺陷）
    if (v := _coerce_positive(spec.timeout_seconds)) is not None: return v, "tool_defined"
    # 3) 智能体版本级 → 4) 全局级
    if (v := parse_agent_version_toolcall_timeout(agent_timeout)) is not None: return v, "agent"
    return parse_agent_max_toolcall_timeout(global_timeout), "global"
